50 {
51 return gamma_sub / m_cmol_sub - gamma_biomass * y_x / m_cmol_biomass;

400 KOKKOS_INLINE_FUNCTION Acetate::uptake_rates
401 Acetate::uptake(const FloatType c_s,
402 const FloatType c_o2,
403 const FloatType c_ac,
404 const FloatType a_max,
405 const FloatType length,
406 const FloatType d_t)
407 {
408 const FloatType phi_s_max_p = a_max * lin_density * inv_y_x_go;
409 const FloatType phi_o2_max_p = f_o2_max * y_o2_go * phi_s_max_p;
410 const FloatType phi_ac_max_p = f_ac_max * phi_s_max_p;
411
412 const FloatType share
413 = (length * lin_density) / ((f_pos * x_c_max * d_t) + d_t_floor);
414
415 // Monod factors, one division for the three
416 const auto [m_s, m_o2, m_ac] = monod_batch(c_s, k_s, c_o2, k_o, c_ac, k_a);
417
418 uptake_rates phi{};
419 phi[S] = phi_s_max_p * m_s;
420 phi[O2] = Kokkos::min(phi_o2_max_p * m_o2, share * c_o2);
421 phi[Ac] = Kokkos::min(phi_ac_max_p * m_ac, share * c_ac);
422 return phi;
423 }

425 KOKKOS_INLINE_FUNCTION Acetate::reaction_rates
426 Acetate::metabolism(const uptake_rates& phi, const FloatType nu_p)
427 {
428 constexpr auto zero = static_cast<FloatType>(0);
429 reaction_rates r = {}; // zero initialised
430
431 const FloatType s_ox = Kokkos::min(phi[S], phi[O2] * inv_y_o2_go);
432 const FloatType go = Kokkos::min(s_ox, nu_p * inv_y_x_go);
433
434 const FloatType growth_res = Kokkos::max(nu_p - go * y_x_go, zero);
435 const FloatType o2_res = Kokkos::max(phi[O2] - go * y_o2_go, zero);
436
437 r[i_go] = go;
438
439 //>0 iff nu_p limits
440 r[i_ov] = s_ox - go;
441
442 //>0 iff o2 limits
443 // phi_s limits -> phis=s_0x, phi_s-s_ox=0
444 // nu limits -> go = nu <phi_s
445 r[i_fe] = Kokkos::min(phi[S] - s_ox, growth_res * inv_y_x_fe);
446
447 //>0 iff gl limits
448 const FloatType ox_growth_res
449 = Kokkos::min(o2_res * inv_y_o2_ao, growth_res * inv_y_x_ao);
450
451 r[i_ao] = Kokkos::min(phi[Ac], ox_growth_res);
452
453 return r;
454 }

490 KOKKOS_INLINE_FUNCTION MC::Status
491 Acetate::update([[maybe_unused]] const MC::pool_type& random_pool,
492 FloatType d_t,
493 std::size_t idx,
494 const SelfParticle& arr,
495 const SelfContribs& arr_contribs,
496 const std::size_t position_index,
497 const MC::LocalConcentration& c)
498 {
499 const uptake_rates phi
500 = uptake(GET_CLAMPED_CONCENTRATION_CAST(FloatType, S),
501 GET_CLAMPED_CONCENTRATION_CAST(FloatType, O2),
502 GET_CLAMPED_CONCENTRATION_CAST(FloatType, Ac),
503 GET_PROPERTY(particle_var::a_max),
504 GET_PROPERTY(particle_var::length),
505 d_t);
506
507 // Growth capacity of the individual, in ]0, a_max] by division()
508 const FloatType nu_p = GET_PROPERTY(particle_var::a_p) * lin_density;
509
510 const reaction_rates r = metabolism(phi, nu_p);
511
512 // Liquid contributions [kg/s] and realised growth nu [kgX/s]
513 const species_rates d = apply_stoichiometry<stoichiometry>(r);
514
515 constexpr FloatType inv_lin_density
516 = static_cast<FloatType>(1) / lin_density;
517 const FloatType a_e = d[X] * inv_lin_density;
518
519 GET_PROPERTY(particle_var::a_e) = a_e;
520 GET_PROPERTY(particle_var::a_go) = r[i_go] * y_x_go * inv_lin_density;
521 GET_PROPERTY(particle_var::a_fe) = r[i_fe] * y_x_fe * inv_lin_density;
522 GET_PROPERTY(particle_var::a_ao) = r[i_ao] * y_x_ao * inv_lin_density;
523
524 GET_PROPERTY(particle_var::length) += d_t * a_e;
525
526 GET_PROPERTY(particle_var::phi_s) = d[S];
527 GET_PROPERTY(particle_var::phi_o2) = d[O2];
528 GET_PROPERTY(particle_var::phi_a) = d[Ac];
529
530 GET_CONTRIBS(S) = d[S];
531 GET_CONTRIBS(O2) = d[O2];
532 GET_CONTRIBS(Ac) = d[Ac];
533 GET_CONTRIBS(Co2) = d[Co2];
534
535 return check_div(GET_PROPERTY(Self::particle_var::length),
536 GET_PROPERTY(Self::particle_var::l_max));
537 }

539 KOKKOS_INLINE_FUNCTION void
540 Acetate::division([[maybe_unused]] const MC::pool_type& random_pool,
541 std::size_t idx,
542 std::size_t idx2,
543 const SelfParticle& arr,
544 const SelfParticle& buffer_arr)
545 {
546 Kokkos::View<FloatType**,
547 ComputeSpace::array_layout,
548 Kokkos::MemoryTraits<Kokkos::MemoryTraitsFlags::Restrict>>
549 buffer_e = buffer_arr;
550
551 const FloatType current_l = GET_PROPERTY(particle_var::length);
552 const FloatType current_a_e = GET_PROPERTY(particle_var::a_e);
553 const FloatType currernt_a_max = GET_PROPERTY(particle_var::a_max);
554 // Local copies: nvcc rejects a static constexpr member passed by reference
555 // to Kokkos::min/max/clamp ("undefined in device code")
556 constexpr FloatType ap_min = a_p_min;
557 static constexpr auto local_l = l_max_dist;
558
559 const FloatType new_current_length = current_l / 2.F;
560 GET_PROPERTY(particle_var::length) = new_current_length;
561 constexpr auto binf = INDEX_FROM_ENUM(particle_var::length);
562 constexpr auto bsup = INDEX_FROM_ENUM(particle_var::a_e);
563 for (auto i = binf; i < bsup; ++i)
564 {
565 COPY_PROPERTY_TO(i, idx2, buffer_e);
566 }
567
568 // Effective elongation of the mother, floored: a fully starved cell would
569 // otherwise give log(0)
570 // const FloatType a_e_i = Kokkos::clamp(current_a_e, ap_min, a_max_i);
571 // Centre of the redistribution. a_e <= a_p always holds, so centring on
572 // a_e alone makes the capacity of a lineage a decreasing sequence that
573 // converges to 0; the pull toward a_max keeps a_p away from 0 and lets a
574 // starved lineage recover when the environment improves.
575 //
576 // constexpr FloatType a_p_recovery = 0.1;
577
578 // const auto a_target = static_cast<double>(
579 // current_a_e
580 // + (a_p_recovery * (GET_PROPERTY(particle_var::a_p) - current_a_e)));
581 //
582 //
583 const FloatType r = 0.2;
584 const auto satisfaction = current_a_e / GET_PROPERTY(particle_var::a_p);
585 const auto a_max = GET_PROPERTY(particle_var::a_max);
586 const auto a_target = static_cast<double>(
587 current_a_e + r * satisfaction * (a_max - current_a_e)
588
589 );
590
591 // const double average = Kokkos::log(a_target) - sigma * sigma / 2;
592
593 const double sigma = Self::sigma_log_inheritance;
594
595 //
596 //
597 // Centre of the redistribution.
598 // The log-normal distribution is centred on a_e, allowing a_p
599 // to naturally follow environmental improvements. Even when
600 // a_e == a_p, the distribution retains a non-zero probability
601 // of producing a higher a_p, allowing gradual lineage recovery.
602 const double average = Kokkos::log(a_target) - sigma * sigma / 2;
603 const auto dist = MC::Distributions::LogNormal<double>(average, sigma);
604
605 auto gen = random_pool.get_state();
606 const auto ap_gen_1 = static_cast<FloatType>(dist.draw(gen));
607 const auto ap_gen_2 = static_cast<FloatType>(dist.draw(gen));
608 const FloatType lmax1 = local_l.draw(gen);
609 const FloatType lmax2 = local_l.draw(gen);
610 random_pool.free_state(gen);
611
612 const FloatType a_up = 1.5 * a_max;
613 // Clamped to [a_p_min, a_max]
614 const auto ap_1 = Kokkos::clamp(ap_gen_1, ap_min, a_up);
615 const auto ap_2 = Kokkos::clamp(ap_gen_2, ap_min, a_up);
616
617 GET_PROPERTY(particle_var::a_p) = ap_1;
618 GET_PROPERTY(particle_var::l_max) = lmax1;
619 GET_PROPERTY_FROM(idx2, buffer_arr, particle_var::a_p) = ap_2;
620 GET_PROPERTY_FROM(idx2, buffer_arr, particle_var::l_max) = lmax2;
621 }
