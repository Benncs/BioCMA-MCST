130 template <typename Tag>
131 KOKKOS_INLINE_FUNCTION void
132 update_block(const TeamMember& team,
133 const std::size_t p0,
134 const std::size_t upper_bound) const
135 {
136 const auto& status = m_particles.status;
137 Kokkos::parallel_for(
138 Kokkos::TeamThreadRange(team, 0, upper_bound),
139 [&](const std::size_t i)
140 {
141 const std::size_t flatten_index = p0 + i;
142 if (status(flatten_index) != MC::Status::Idle)
143 {
144 return;
145 }
146 if constexpr (std::is_same_v<Tag, Tag0D>)
147 {
148 // Position is always 0 in 0D
149 update_particle(flatten_index, 0);
150 }
151 else
152 {
153 update_particle(flatten_index, m_particles.position(flatten_index));
154 }
155 });
156 }

158 KOKKOS_INLINE_FUNCTION
159 void
160 operator()(Tag0D _tag, const TeamMember& team) const
161 {
162 (void)_tag;
163 const std::size_t p0 = team.league_rank() * m_particle_per_team;
164 const auto upper_bound = team_upper_bound(team);
165 const auto& contribs = m_particles.contribs;
166 const auto& status = m_particles.status;
167
168 update_block<Tag0D>(team, p0, upper_bound);
169
170 team.team_barrier();
171
172 // One iteration per species n_c is typicall in [2,10]
173 // n_c*upper_bound itreation but less atomic, less contention
174 for (std::size_t j = 0; j < M::n_c; ++j)
175 {
176 float_type sum = 0;
177 Kokkos::parallel_reduce(
178 Kokkos::TeamThreadRange(team, 0, upper_bound),
179 [&](const std::size_t i, float_type& local_sum)
180 {
181 const std::size_t flatten_index = p0 + i;
182 if (status(flatten_index) == MC::Status::Idle)
183 {
184 local_sum += static_cast<float_type>(
185 m_particles.get_weight(flatten_index)
186 * contribs(flatten_index, j));
187 }
188 },
189 sum);
190
191 Kokkos::single(Kokkos::PerTeam(team),
192 [&]()
193 {
194 auto access = m_contribution_scatter.access();
195 access(j, 0) += sum;
196 });
197 }
198 }

200 KOKKOS_INLINE_FUNCTION
201 void
202 operator()(const Tag3D _tag, const TeamMember& team) const
203 {
204 (void)_tag;
205
206 const std::size_t p0 = team.league_rank() * m_particle_per_team;
207 const std::size_t p = team.team_size();
208 const auto upper_bound = team_upper_bound(team);
209 const std::size_t m = (upper_bound + p - 1) / p;
210
211 const auto& positions = m_particles.position;
212 const auto& contribs = m_particles.contribs;
213 const auto& status = m_particles.status;
214
215 update_block<Tag3D>(team, p0, upper_bound);
216
217 team.team_barrier();
218
219 // Stategy here is to reduce contention to acess (scatterview)
220 // One team performs m iteration with one access
221 Kokkos::parallel_for(
222 Kokkos::TeamThreadRange(team, 0, p),
223 [&](const std::size_t tid)
224 {
225 auto access = m_contribution_scatter.access();
226 for (std::size_t k = 0; k < m; ++k)
227 {
228 const std::size_t base = tid + k * p; // stride p
229 const std::size_t flat_index = p0 + base;
230 if (base >= upper_bound)
231 {
232 break;
233 }
234 if (status(flat_index) != MC::Status::Idle)
235 {
236 continue;
237 }
238
239 const double weight = m_particles.get_weight(flat_index);
240 const auto pos = positions(flat_index);
241 Kokkos::parallel_for(
242 Kokkos::ThreadVectorRange(team, 0, M::n_c),
243 [&](const int j)
244 { access(j, pos) += weight * contribs(flat_index, j); });
245 }
246 });
247 }
