79 template <ModelType M>
80 std::vector<double>
81 mean_properties(const MC::ParticlesContainer<M>& container)
82 {
83
84 Kokkos::View<double[M::n_var]> means("means_properties");
85 std::vector<double> vec_means;
86 vec_means.reserve(M::n_var);
87
88 double denum = 0.;
89 const double overflow_limit = 1e30;
90 auto model = container.model;
91
92 auto policy = Kokkos::MDRangePolicy<ComputeSpace, Kokkos::Rank<2> >(
93 { 0, 0 }, { container.n_particles(), M::n_var });
94
95 Kokkos::parallel_reduce(
96 "total_weights",
97 Kokkos::RangePolicy<>(0, container.n_particles()),
98 KOKKOS_LAMBDA(const int i, double& red) {
99 red += container.get_weight(i) / overflow_limit;
100 },
101 denum);
102
103 Kokkos::parallel_for(
104 "mc_means", policy, KOKKOS_LAMBDA(const int i, const int j) {
105 const double w = container.get_weight(i);
106 const auto v = model(i, j);
107 Kokkos::atomic_add(&means(j), v * w);
108 });
109
110 const auto tmp = Kokkos::create_mirror_view_and_copy(HostSpace(), means);
111 for (int i = 0; i < M::n_var; ++i)
112 {
113 double real_mean = tmp(i) / (overflow_limit * denum);
114 vec_means.push_back(real_mean);
115 }
116
117 return vec_means;
118 }
