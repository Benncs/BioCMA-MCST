68 KOKKOS_INLINE_FUNCTION std::size_t
69 __find_next_compartment(
70 const bool do_serch,
71 const MC::NeighborsView<ComputeSpace, true>& neighbors,
72 const MC::CumulativeProbabilityView<ComputeSpace, true>&
73 cumulative_probability,
74 const std::size_t i_compartment,
75 const double random_number)
76 {
77 const int mask_do_serch = static_cast<int>(do_serch);
78 const int max_neighbor = static_cast<int>(neighbors.extent(1));
79
80 KOKKOS_ASSERT(max_neighbor >= 1);
81 KOKKOS_ASSERT(random_number <= 1. && random_number >= 0.);
82 KOKKOS_ASSERT(neighbors.extent(1) == cumulative_probability.extent(1));
83
84 // Do not use cumulative_probability(i_compartment, max_neighbor - 1)==1
85 // Bcause proba can be 0.999999999
86 // KOKKOS_ASSERT(cumulative_probability(i_compartment, max_neighbor -
87 // 1)>0.95); //5% relative
88
89 // v1.1: This assert was there to be sure that the particle will leave but
90 // actually
91 // not needed because if rand< then left will be the last at the the ned of
92 // the loop
93 // KOKKOS_ASSERT(cumulative_probability(i_compartment, max_neighbor - 1)
94 // >= random_number);
95
96 int left = 0;
97 int right = mask_do_serch * (max_neighbor - 1);
98 while (left < right)
99 {
100 const int mid = (left + right) >> 1; // NOLINT
101 const auto pm = cumulative_probability(i_compartment, mid);
102 const int mask = static_cast<int>(random_number > pm);
103 left = mask * (mid + 1) + (1 - mask) * left;
104 right = mask * right + (1 - mask) * mid;
105 }
106 KOKKOS_ASSERT(left >= 0 && static_cast<size_t>(left) < neighbors.extent(1));
107 const auto ret = i_compartment * (1 - mask_do_serch)
108 + neighbors(i_compartment, left) * mask_do_serch;
109 return ret;
110 }

112 template <typename ViewType1>
113 KOKKOS_INLINE_FUNCTION void
114 find_flow(const ViewType1& leaving_flow,
115 const std::size_t position,
116 MC::LeavingFlow::float_type& val_flow,
117 MC::LeavingFlow::float_type& _liquid_volume)
118 {
119 std::size_t i_flow = 0;
120 const std::size_t n_flow = leaving_flow.size();
121 val_flow = 0.;
122 _liquid_volume = 0.;
123 // do-while because n_flow is likely to be 1
124 do
125 {
126 const auto& [index, flow, liquid_volume] = leaving_flow(i_flow++);
127 if (position == index)
128 {
129 val_flow = flow;
130 _liquid_volume = liquid_volume;
131 break;
132 }
133 } while (i_flow < n_flow);
134 }

211 KOKKOS_INLINE_FUNCTION void
212 operator()(TagMove /*tag*/,
213 const Kokkos::TeamPolicy<ComputeSpace>::member_type& team) const
214 {
215 using ScratchSpace
216 = Kokkos::TeamPolicy<>::execution_space::scratch_memory_space;
217 using ScratchView = Kokkos::View<float*, ScratchSpace>;
218
219 const std::size_t count = m_p_team_move;
220 const std::size_t p0 = team.league_rank() * count;
221 const std::size_t n_particle = n_particles;
222
223 const auto upper_bound
224 = ((p0 + count) >= n_particle) ? n_particle - p0 : count;
225 KOKKOS_ASSERT(upper_bound > 0 && upper_bound < n_particle);
226 const auto& rp = random_pool;
227
228 const std::size_t N = count * 2;
229
230 // rng is actually a flat m*p array
231 const std::size_t p = team.team_size();
232 const std::size_t m = (N + p - 1) / p;
233
234 ScratchView rng(team.team_scratch(0), N);
235
236 // Use "tiling" to minimize contention when aquired_state
237 // State is aquired p times instead of N, it is supposed to reduce
238 // contention
239
240 Kokkos::parallel_for(Kokkos::TeamThreadRange(team, 0, p),
241 [&rp, &rng, N, m, p](const std::size_t idx)
242 {
243 // Ok to use here, get_state should be called in
244 // each thread
245 auto gen = rp.get_state();
246
247 // current thread iteration p times with the same
248 // state
249 for (std::size_t k = 0; k < m; ++k)
250 {
251 const std::size_t i = idx + k * p; // stride p
252
253 if (i >= N)
254 {
255 break;
256 }
257
258 rng(i) = gen.frand(0., 1.);
259 }
260
261 rp.free_state(gen);
262 });
263 team.team_barrier();
264
265 // We can use flat array index here ordering of random doesnt matter
266 // Move events are accumulated per team and flushed with a single atomic,
267 // same strategy as the Exit tally in TagLeave/TagLeaveB0D below.
268 std::size_t team_move_count = 0;
269 Kokkos::parallel_reduce(
270 Kokkos::TeamThreadRange(team, 0, upper_bound),
271 [&](const std::size_t idx, std::size_t& thread_move_count)
272 {
273 const auto flat_index = p0 + idx;
274
275 if (status(flat_index) == MC::Status::Idle)
276 {
277
278 const std::size_t base = idx * 2;
279 KOKKOS_ASSERT(base + 1 < N);
280 const auto rng1 = rng(base);
281 const auto rng2 = rng(base + 1);
282 if (handle_move(flat_index, rng1, rng2))
283 {
284 ++thread_move_count;
285 }
286 }
287 },
288 team_move_count);
289
290 if constexpr (AutoGenerated::FlagCompileTime::enable_event_counter)
291 {
292 team.team_barrier();
293 Kokkos::single(
294 Kokkos::PerTeam(team),
295 [&]() { events.add<MC::EventType::Move>(team_move_count); });
296 }
297 }

347 KOKKOS_INLINE_FUNCTION void
348 operator()(TagMoveLeave /*tag*/,
349 const Kokkos::TeamPolicy<ComputeSpace>::member_type& team,
350 std::size_t& local_dead_count) const
351 {
352 const std::size_t N = m_p_team_move;
353 const std::size_t p0 = team.league_rank() * N;
354 const auto upper_bound
355 = ((p0 + N) >= n_particles) ? n_particles - p0 : N;
356
357 KOKKOS_ASSERT(upper_bound > 0 && upper_bound <= n_particles);
358
359 const auto& rp = random_pool;
360 const auto& lf = move.leaving_flow;
361
362 const std::size_t p = team.team_size();
363 const std::size_t m = (upper_bound + p - 1) / p;
364
365 MoveLeaveTally team_tally{ 0, 0 };
366
367 Kokkos::parallel_reduce(
368 Kokkos::TeamThreadRange(team, 0, p),
369 [&](const std::size_t tid, MoveLeaveTally& acc)
370 {
371 auto gen = rp.get_state();
372
373 for (std::size_t k = 0; k < m; ++k)
374 {
375 const std::size_t idx = tid + k * p; // stride p
376
377 if (idx >= upper_bound)
378 {
379 break;
380 }
381 const std::size_t flat_index = p0 + idx;
382
383 // Single read of the position, reused by the exit test below
384 std::size_t position = positions(flat_index);
385
386 if (status(flat_index) == MC::Status::Idle)
387 {
388 bool moved = false;
389 position = next_compartment(position,
390 gen.frand(0.F, 1.F),
391 gen.frand(0.F, 1.F),
392 moved);
393 positions(flat_index) = position;
394 acc.moved += static_cast<std::size_t>(moved);
395 }
396
397 ages(flat_index, 0) += d_t;
398 handle_exit_at(flat_index, lf, position, gen, acc.dead);
399 }
400
401 rp.free_state(gen);
402 },
403 team_tally);
404
405 team.team_barrier();
406
407 Kokkos::single(
408 Kokkos::PerTeam(team),
409 [&]()
410 {
411 if constexpr (AutoGenerated::FlagCompileTime::enable_event_counter)
412 {
413 events.add<MC::EventType::Move>(team_tally.moved);
414 events.add<MC::EventType::Exit>(team_tally.dead);
415 }
416 local_dead_count += team_tally.dead;
417 });
418 }

432 KOKKOS_INLINE_FUNCTION void
433 operator()(TagLeaveB0D _tag,
434 const Kokkos::TeamPolicy<ComputeSpace>::member_type& team_handle,
435 std::size_t& local_dead_count) const
436 {
437
438 (void)_tag;
439
440 const auto& lf = move.leaving_flow;
441
442 // In 0d, the flow is the same for every particle
443 // Only one member of the team calculate lambda and
444 // broadcast it to other members
445 double lambda = 0.;
446 Kokkos::single(
447 Kokkos::PerTeam(team_handle),
448 [&lf, dt = this->d_t](double& local_lambda)
449 {
450 const auto& [_, flow_value, liquid_volume] = lf(0);
451
452 local_lambda = dt * flow_value / liquid_volume;
453 },
454 lambda);
455
456 team_handle.team_barrier();
457 // In 0d context we can even be sure that flow will never be 0
458 // Note: if flow is 0, proba will be 0 and then the result will be the
459 // same Can we remove the condition?
460 //
461 if (lambda == 0.)
462 {
463 return;
464 }
465 const std::size_t N = m_p_team_leave;
466 const std::size_t p0 = team_handle.league_rank() * N;
467 const auto upper_bound = ((p0 + N) >= n_particles) ? n_particles - p0 : N;
468
469 KOKKOS_ASSERT(upper_bound > 0 && upper_bound <= n_particles);
470
471 const auto& rp = random_pool;
472
473 const std::size_t p = team_handle.team_size();
474
475 const std::size_t m = (upper_bound + p - 1) / p;
476
477 std::size_t team_dead_count = 0;
478
479 // For each thread of the team we assigm m iterations
480 // This divide the number of call of get_state by m and may reduce
481 // contention especially on CPU typically P = 1 on cpu then m is
482 // equal to npt (128-4096)
483 Kokkos::parallel_reduce(
484 Kokkos::TeamThreadRange(team_handle, 0, p),
485 [&](const std::size_t tid, std::size_t& dead_count)
486 {
487 auto gen = rp.get_state();
488
489 for (std::size_t k = 0; k < m; ++k)
490 {
491 const std::size_t idx = tid + k * p; // stride p
492
493 if (idx >= upper_bound)
494 {
495 break;
496 }
497 const std::size_t flat_index = p0 + idx;
498
499 if (status(flat_index) != MC::Status::Idle)
500 {
501 continue;
502 }
503 const double r = gen.drand(0., 1.);
504
505 perform_exit(
506 probability_leaving<decltype(r), precision_tag>(r, lambda),
507 flat_index,
508 dead_count);
509 }
510 rp.free_state(gen);
511 },
512 team_dead_count);
513
514 team_handle.team_barrier();
515
516 Kokkos::single(
517 Kokkos::PerTeam(team_handle),
518 [&]()
519 {
520 if constexpr (AutoGenerated::FlagCompileTime::enable_event_counter)
521 {
522 events.add<MC::EventType::Exit>(team_dead_count);
523 }
524 local_dead_count += team_dead_count;
525 });
526 }

532 KOKKOS_FORCEINLINE_FUNCTION std::size_t
533 next_compartment(const std::size_t i_current_compartment,
534 const float rng1,
535 const float rng2,
536 bool& moved) const
537 {
538 KOKKOS_ASSERT(rng1 >= 0. && rng1 <= 1 && rng2 >= 0. && rng2 <= 1);
539 KOKKOS_ASSERT(
540 i_current_compartment < move.liquid_volume.extent(0)
541 && "Particle position is incorect (greater than compartment number)");
542
543 moved = probability_leaving<float, fast_tag>(
544 rng1,
545 move.liquid_volume(i_current_compartment),
546 move.diag_transition(i_current_compartment),
547 d_t);
548
549 const auto next = __find_next_compartment(moved,
550 move.neighbors,
551 move.cumulative_probability,
552 i_current_compartment,
553 rng2);
554 KOKKOS_ASSERT(
555 next < move.liquid_volume.extent(0)
556 && " Position after move is greater than compartment number");
557 return next;
558 }

601 template <typename ExecSpace>
602 KOKKOS_FORCEINLINE_FUNCTION std::size_t
603 handle_exit(
604 const std::size_t idx,
605 const Kokkos::View<const MC::LeavingFlow*, ExecSpace>& leaving_flow,
606 std::size_t& dead_count) const
607 {
608
609 const std::size_t position = positions(idx);
610 // Strategy:
611 // first find the value of leaving flow (0-> particle doesn´t leave)
612 // do-while +early break is ok as n_flow is likely <10
613 // second: calculate probability leaving, flow=0 => p=0 theres no need
614 // to check condition
615 MC::LeavingFlow::float_type found_flow_value = 0.;
616 MC::LeavingFlow::float_type found_liquid_volume = 0.;
617 find_flow(leaving_flow, position, found_flow_value, found_liquid_volume);
618
619 // Cases
620 // only for few particles
621 if (found_flow_value != 0.)
622 {
623
624 auto gen = random_pool.get_state();
625 const auto r = gen.frand(0., 1.);
626 random_pool.free_state(gen);
627
628 KOKKOS_ASSERT(found_liquid_volume > 0.);
629 KOKKOS_ASSERT(found_flow_value > 0.);
630 const bool p = probability_leaving<decltype(r), precision_tag>(
631 r, found_liquid_volume, found_flow_value, d_t);
632 perform_exit(p, idx, dead_count);
633 }
634
635 return 0;
636 }
