173 template <typename T, std::size_t N>
174 KOKKOS_INLINE_FUNCTION constexpr Kokkos::Array<T, N>
175 batch_reciprocal(const Kokkos::Array<T, N>& x)
176 {
177 static_assert(N > 0, "batch_reciprocal needs at least one value");
178
179 Kokkos::Array<T, N> prefix{};
180 prefix[0] = x[0];
181 Common::static_for<N - 1>(
182 [&](auto k)
183 {
184 constexpr std::size_t i = decltype(k)::value + 1;
185 prefix[i] = prefix[i - 1] * x[i];
186 });
187
188 T inv = static_cast<T>(1) / prefix[N - 1];
189
190 Kokkos::Array<T, N> out{};
191 Common::static_for<N - 1>(
192 [&](auto k)
193 {
194 constexpr std::size_t i = N - 1 - decltype(k)::value;
195 out[i] = inv * prefix[i - 1];
196 inv *= x[i];
197 });
198 out[0] = inv;
199
200 return out;
201 }
202 } // namespace impl

216 template <typename... Ts>
217 KOKKOS_INLINE_FUNCTION constexpr auto
218 monod_batch(Ts... args)
219 {
220 using T = std::common_type_t<Ts...>;
221 static_assert(FloatingPointType<T>, "monod_batch: floating point only");
222 static_assert(sizeof...(Ts) > 0 && sizeof...(Ts) % 2 == 0,
223 "monod_batch: arguments are (c, k) pairs");
224 constexpr std::size_t n = sizeof...(Ts) / 2;
225
226 const Kokkos::Array<T, 2 * n> ck = { { static_cast<T>(args)... } };
227
228 Kokkos::Array<T, n> denominators{};
229 Common::static_for<n>(
230 [&](auto i)
231 {
232 constexpr std::size_t j = decltype(i)::value;
233 denominators[j] = ck[2 * j] + ck[(2 * j) + 1];
234 });
235
236 const auto inv = impl::batch_reciprocal(denominators);
237
238 Kokkos::Array<T, n> m{};
239 Common::static_for<n>(
240 [&](auto i)
241 {
242 constexpr std::size_t j = decltype(i)::value;
243 m[j] = ck[2 * j] * inv[j];
244 });
245 return m;
246 }

260 template <auto Stoichiometry, typename Rates>
261 KOKKOS_INLINE_FUNCTION constexpr auto
262 apply_stoichiometry(const Rates& r)
263 {
264 // clang sees the template parameter object as a const reference
265 using matrix_type = std::remove_cvref_t<decltype(Stoichiometry)>;
266 using row_type = typename matrix_type::value_type;
267 using T = typename row_type::value_type;
268 constexpr std::size_t n_rows = matrix_type::size();
269 constexpr std::size_t n_reactions = row_type::size();
270 static_assert(n_reactions == Rates::size(),
271 "apply_stoichiometry: one rate per matrix column");
272
273 Kokkos::Array<T, n_rows> d{};
274 Common::static_for<n_rows>(
275 [&](auto j)
276 {
277 Common::static_for<n_reactions>(
278 [&](auto k)
279 {
280 // Local copies: nvcc rejects integral_constant::value used as
281 // a runtime subscript ("undefined in device code")
282 constexpr std::size_t row = decltype(j)::value;
283 constexpr std::size_t col = decltype(k)::value;
284 constexpr T n_jk = Stoichiometry[row][col];
285 if constexpr (n_jk != 0)
286 {
287 d[row] += n_jk * r[col];
288 }
289 });
290 });
291 return d;
292 }
